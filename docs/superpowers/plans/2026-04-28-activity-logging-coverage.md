# Activity Logging Coverage Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire comprehensive activity logging across the curated audit set so the Activity screen reflects real CRUD instead of staying empty.

**Architecture:** A `logActivity` helper writes one `ActivityLog` row per mutation, called explicitly from each covered API route after the mutation succeeds. A Prisma migration adds a proper `userId` FK plus two composite indexes. A permission swap moves the read route from `sales.view` to `activity.view` (Owner/Admin only). Bulk endpoints log a single summary row capped to 100 affected IDs.

**Tech Stack:** Next.js 16 API routes, Prisma 7 (PostgreSQL), Vitest, NextAuth v5 sessions, TypeScript.

**Spec:** `docs/superpowers/specs/2026-04-28-activity-logging-coverage-design.md`

```ts
import { prisma } from './prisma';

export type ActivityAction =
  | 'create' | 'update' | 'delete'
  | 'bulk_update' | 'bulk_delete' | 'bulk_giveaway' | 'bulk_tag' | 'bulk_msrp'
  | 'merge' | 'import' | 'pack' | 'verify'
  | 'favorite_set' | 'note_add' | 'note_delete'
  | 'rule_run' | 'sync_run' | 'permissions_update';

export type ActivityResource =
  | 'item' | 'product' | 'purchase' | 'expense' | 'expense_category'
  | 'customer' | 'show' | 'consignor' | 'consignment' | 'consignor_payout'
  | 'rule' | 'alert' | 'tag' | 'shipment' | 'message_template'
  | 'user' | 'setting' | 'sync_config';

export interface LogActivityInput {
  tenantId: string;
  userId: number;
  resourceType: ActivityResource;
  resourceId: string;
  action: ActivityAction;
  oldValues?: Record<string, unknown> | null;
  newValues?: Record<string, unknown> | null;
}

function shortName(name: string | null | undefined, email: string | null | undefined): string {
  if (name && name.trim()) {
    const parts = name.trim().split(/\s+/);
    if (parts.length === 1) return parts[0];
    return `${parts[0]} ${parts[parts.length - 1][0]}.`;
  }
  return email ?? 'unknown';
}

export async function logActivity(input: LogActivityInput): Promise<void> {
  try {
    const user = await prisma.user.findUnique({
      where: { id: input.userId },
      select: { name: true, email: true },
    });
    const userInfo = shortName(user?.name, user?.email);

    await prisma.activityLog.create({
      data: {
        tenantId: input.tenantId,
        userId: input.userId,
        resourceType: input.resourceType,
        resourceId: input.resourceId,
        action: input.action,
        oldValues: input.oldValues ?? null,
        newValues: input.newValues ?? null,
        userInfo,
      },
    });
  } catch (err) {
    console.error('[activity] failed to write log', err, {
      tenantId: input.tenantId,
      userId: input.userId,
      resourceType: input.resourceType,
      action: input.action,
    });
  }
}
```

```ts
function pickFields(row: Record<string, unknown>, fields: string[]): Record<string, unknown> {
  const out: Record<string, unknown> = {};
  for (const f of fields) {
    if (Object.prototype.hasOwnProperty.call(row, f)) {

  }
  return out;
```

```ts
import { NextRequest, NextResponse } from "next/server";
import { getTenantContext, requirePermission, handleAuthError } from "@/lib/tenant";
import { prisma } from "@/lib/prisma";
import { logActivity } from "@/lib/activity";

export async function PATCH(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  try {
    const ctx = await getTenantContext(req);
    requirePermission(ctx, 'customers.edit');

    const { id } = await params;
    const customerId = parseInt(id);

    const customer = await prisma.customer.findFirst({
      where: { id: customerId, tenantId: ctx.tenantId },
      select: { id: true, username: true },
    });

    if (!customer) {
      return NextResponse.json(
        { error: "Customer not found" },
        { status: 404 }
      );
    }

    const existingFavorite = await prisma.activityLog.findFirst({
      where: {
        tenantId: ctx.tenantId,
        resourceType: "customer",
        resourceId: String(customerId),
        action: "favorite_set",
        userId: ctx.userId,
      },
      orderBy: { createdAt: "desc" },
    });

    const currentlyFavorited = existingFavorite
      ? (existingFavorite.newValues as Record<string, unknown>)?.isFavorite === true
      : false;

    const newFavoriteState = !currentlyFavorited;

    await logActivity({
      tenantId: ctx.tenantId,
      userId: ctx.userId,
      resourceType: 'customer',
      resourceId: String(customerId),
      action: 'favorite_set',
      oldValues: { isFavorite: currentlyFavorited },
      newValues: { isFavorite: newFavoriteState },
    });

    return NextResponse.json({
      success: true,
      data: {
        customerId,
        isFavorite: newFavoriteState,
      },
    });
  } catch (error) {
    return handleAuthError(error);
  }
}
```
